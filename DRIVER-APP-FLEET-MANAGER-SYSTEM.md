## 💾 DATABASE SCHEMA

```sql
-- Drivers Table
CREATE TABLE drivers (
    driver_id INT AUTO_INCREMENT PRIMARY KEY,
    driver_code VARCHAR(20) UNIQUE NOT NULL,    -- DRV-042
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    line_id VARCHAR(50),
    email VARCHAR(100),
    license_number VARCHAR(50),
    license_expiry DATE,
    status VARCHAR(20) DEFAULT 'active',        -- active, on_duty, off_duty, on_leave, terminated
    
    -- Performance Metrics
    total_trips INT DEFAULT 0,
    total_distance DECIMAL(10,2) DEFAULT 0,
    total_revenue DECIMAL(12,2) DEFAULT 0,
    acceptance_rate DECIMAL(5,2),               -- %
    rejection_count INT DEFAULT 0,
    on_time_percentage DECIMAL(5,2),            -- %
    safety_score DECIMAL(5,2),                  -- 0-100
    customer_rating DECIMAL(3,2),               -- 0-5.0
    performance_score DECIMAL(5,2),             -- 0-100
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Jobs Table
CREATE TABLE jobs (
    job_id INT AUTO_INCREMENT PRIMARY KEY,
    rn_number VARCHAR(50) UNIQUE NOT NULL,
    cb_number VARCHAR(50),
    container_number VARCHAR(11),
    
    -- Assignment
    assigned_to_driver_id INT,
    assigned_to_truck_id INT,
    assigned_at TIMESTAMP,
    assigned_by INT,                            -- User ID
    
    -- Status
    status VARCHAR(20) DEFAULT 'pending',       -- pending, assigned, confirmed, in_progress, completed, cancelled
    
    -- Timeline
    confirmed_at TIMESTAMP,
    response_time_seconds INT,                  -- เวลายืนยันรับงาน (วินาที)
    started_at TIMESTAMP,
    completed_at TIMESTAMP,
    
    -- Rejection
    rejected_at TIMESTAMP,
    rejection_reason TEXT,
    rejection_count INT DEFAULT 0,
    
    -- Performance
    on_time BOOLEAN,
    delay_minutes INT,
    customer_rating DECIMAL(3,2),
    
    -- Revenue
    driver_payment DECIMAL(10,2),
    bonuses JSON,                               -- {safety: 2000, on_time: 1000}
    
    FOREIGN KEY (assigned_to_driver_id) REFERENCES drivers(driver_id)
);

-- GPS Tracking Table
CREATE TABLE gps_tracking (
    tracking_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL,
    driver_id INT NOT NULL,
    
    latitude DECIMAL(10,7),
    longitude DECIMAL(10,7),
    speed DECIMAL(5,2),                         -- km/h
    heading INT,                                -- 0-360 degrees
    accuracy DECIMAL(6,2),                      -- meters
    battery_level INT,                          -- %
    
    timestamp TIMESTAMP NOT NULL,
    
    FOREIGN KEY (job_id) REFERENCES jobs(job_id),
    FOREIGN KEY (driver_id) REFERENCES drivers(driver_id),
    
    INDEX idx_job (job_id),
    INDEX idx_timestamp (timestamp DESC)
);

-- Temperature Logs Table (Reefer)
CREATE TABLE temperature_logs (
    log_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL,
    container_number VARCHAR(11),
    
    set_temp DECIMAL(4,1),
    supply_temp DECIMAL(4,1),
    return_temp DECIMAL(4,1),
    humidity DECIMAL(5,2),
    ventilation_percent INT,
    alarm_status VARCHAR(20),
    
    latitude DECIMAL(10,7),
    longitude DECIMAL(10,7),
    timestamp TIMESTAMP NOT NULL,
    
    FOREIGN KEY (job_id) REFERENCES jobs(job_id),
    INDEX idx_job (job_id),
    INDEX idx_timestamp (timestamp DESC)
);

-- Job Events Table (Timeline)
CREATE TABLE job_events (
    event_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    job_id INT NOT NULL,
    
    event_type VARCHAR(50) NOT NULL,           -- assigned, confirmed, arrived_warehouse, loading_started, departed, etc.
    event_timestamp TIMESTAMP NOT NULL,
    
    latitude DECIMAL(10,7),
    longitude DECIMAL(10,7),
    notes TEXT,
    photos JSON,                                -- Array of photo URLs
    
    FOREIGN KEY (job_id) REFERENCES jobs(job_id),
    INDEX idx_job (job_id),
    INDEX idx_type (event_type)
);

-- Maintenance Requests (from Driver)
CREATE TABLE driver_maintenance_requests (
    request_id INT AUTO_INCREMENT PRIMARY KEY,
    truck_id INT NOT NULL,
    driver_id INT NOT NULL,
    job_id INT,                                 -- ถ้าเกิดระหว่างงาน
    
    urgency VARCHAR(20) NOT NULL,              -- normal, urgent, emergency
    issue_type VARCHAR(50),                     -- engine, brake, tire, lights, etc.
    description TEXT NOT NULL,
    impact VARCHAR(50),                         -- can_drive, caution, limited_speed, cannot_drive
    
    location VARCHAR(255),
    latitude DECIMAL(10,7),
    longitude DECIMAL(10,7),
    
    photos JSON,
    audio_url VARCHAR(255),
    
    status VARCHAR(20) DEFAULT 'pending',       -- pending, acknowledged, in_repair, completed
    
    reported_at TIMESTAMP NOT NULL,
    acknowledged_at TIMESTAMP,
    
    FOREIGN KEY (driver_id) REFERENCES drivers(driver_id)
);
```
